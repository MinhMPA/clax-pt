\documentclass[tikz,border=2pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, calc, fit, backgrounds}

% Palette
\definecolor{shared}{HTML}{E8EFF5}     % shared pipeline modules
\definecolor{cmbmod}{HTML}{F8D7DA}     % CMB branch
\definecolor{galmod}{HTML}{E2D4F0}     % galaxy branch
\definecolor{outputbg}{HTML}{F5F5F5}   % output boxes

\begin{document}
\begin{tikzpicture}[
    every node/.style={font=\sffamily},
    mod/.style={
        rectangle, rounded corners=2pt, draw=#1!60!black!40, fill=#1,
        thick, minimum width=1.6cm, minimum height=0.8cm, align=center,
        font=\sffamily\small
    },
    mod/.default=shared,
    outputbox/.style={
        rectangle, rounded corners=2pt, draw=black!20, fill=outputbg,
        minimum width=1.4cm, minimum height=0.55cm, align=center,
        font=\sffamily\small
    },
    flow/.style={-{Stealth[length=4pt, width=3pt]}, thick, black!55},
    dot/.style={circle, fill=black!55, inner sep=1.2pt},
]

% ===== Row 1: Shared pipeline =====
\node[mod, draw=black!50, fill=white, font=\sffamily\small\bfseries]
    (params) {CosmoParams};

\node[mod, right=0.55cm of params] (bg) {Background};
\node[mod, right=0.55cm of bg] (th) {Thermo};
\node[mod, right=0.55cm of th, minimum width=2.0cm] (eb) {Einstein--Boltzmann};
\node[mod, right=0.55cm of eb] (pr) {Primordial};

% Row 1 arrows
\draw[flow] (params) -- (bg);
\draw[flow] (bg) -- (th);
\draw[flow] (th) -- (eb);
\draw[flow] (eb) -- (pr);

% ===== Fork point below EB =====
\node[dot, below=0.7cm of eb] (fork) {};
\draw[flow] (eb) -- (fork);

% ===== Row 2 left: CMB branch =====
\node[mod=cmbmod, below left=1.5cm and 1.8cm of eb] (tr) {Transfer};
\node[mod=cmbmod, right=0.5cm of tr] (hr) {Harmonic};
\node[mod=cmbmod, right=0.5cm of hr] (le) {Lensing};

% CMB branch arrows
\draw[flow] (fork) -| (tr);
\draw[flow] (tr) -- (hr);
\draw[flow] (hr) -- (le);

% ===== Row 2 right: Galaxy branch =====
\node[mod=galmod, below right=1.5cm and 1.8cm of eb] (ept) {One-loop PT};

% Galaxy branch arrow
\draw[flow] (fork) -| (ept);

% ===== Primordial feeds both branches =====
% Primordial → Transfer (goes down then left)
\draw[flow] (pr.south) -- ++(0,-0.4) -| ([xshift=3pt]tr.north east);

% Primordial → One-loop PT (goes down then right)
% This line crosses the fork→EPT vertical, so we add a bridge/hop
\coordinate (pr_down) at ($(pr.south) + (0,-0.4)$);
\coordinate (ept_top_left) at ([xshift=-3pt]ept.north west);
% The horizontal segment from pr_down going right to ept crosses
% the fork→ept vertical. Compute the crossing point:
\coordinate (cross_x) at ($(fork)!(pr_down)!(ept_top_left)$);
\path let \p1=(fork), \p2=(pr_down) in coordinate (bridge_pt) at (\x1, \y2);
% Draw with a small arc bridge at the crossing
\draw[flow] (pr_down) -- ++(0.3,0)
    -- ([xshift=-0.15cm]bridge_pt)
    arc[start angle=180, end angle=0, radius=0.15cm]
    -- (ept_top_left |- pr_down)
    -- (ept_top_left);

% ===== Row 3: Outputs =====
\node[outputbox, below=0.7cm of tr] (pkout) {$P(k)$};
\node[outputbox, below=0.7cm of le] (clout) {$C_\ell^{TT/EE/TE/BB}$\\[-2pt]{\small lensed\,$C_\ell$}};
\node[outputbox, below=0.7cm of ept] (pgout) {$P_{gg}(k,\ell)$\\[-2pt]{\small $P_{mm},\,P_{gm}$}};

\draw[flow] (tr) -- (pkout);
\draw[flow] (le) -- (clout);
\draw[flow] (ept) -- (pgout);

% ===== Background shading =====
\begin{scope}[on background layer]
    % clax region
    \node[fit=(bg)(th)(eb)(pr)(tr)(hr)(le)(pkout)(clout),
          draw=black!8, fill=shared!30, rounded corners=5pt,
          inner sep=10pt,
          label={[font=\sffamily\footnotesize\bfseries, text=black!30,
                  anchor=north west]north west:\textsc{clax}}] {};
    % clax-pt region
    \node[fit=(ept)(pgout),
          draw=galmod!40, fill=galmod!15, rounded corners=5pt,
          inner sep=10pt,
          label={[font=\sffamily\footnotesize\bfseries, text=black!30,
                  anchor=north east]north east:\textsc{clax-pt}}] {};
\end{scope}

\end{tikzpicture}
\end{document}
